    85→export function Topbar() {
    86→  const { user, setUser, view, setView, setAppMode } = useAppStore();
    87→  const locale = useI18nStore((s) => s.locale);
    88→  const setLocale = useI18nStore((s) => s.setLocale);
    89→  const themeConfig = useThemeCustomStore((s) => s.config);
    90→  const setThemeConfig = useThemeCustomStore((s) => s.setConfig);
    91→  const { resolvedTheme, setTheme } = useTheme();
    92→  const searchOpen = useSearchStore((s) => s.open);
    93→  const admin = isAdmin(user);
    94→  const superAdmin = isSuperAdmin(user);
    95→
    96→  // Real-time notifications state
    97→  const [notifications, setNotifications] = React.useState<NotifItem[]>([]);
    98→  const [unreadTotal, setUnreadTotal] = React.useState(0);
    99→  const [notifOpen, setNotifOpen] = React.useState(false);
   100→  // D-4: brief pulse on the bell when a new notification arrives via WS.
   101→  // Cleared after ~1.2s by the timeout in `flashBell`.
   102→  const [bellPulse, setBellPulse] = React.useState(false);
   103→  const pulseTimer = React.useRef<ReturnType<typeof setTimeout> | null>(null);
   104→
   105→  const loadNotifications = React.useCallback(() => {
   106→    if (!user) return;
   107→    fetch("/api/notifications?unreadOnly=true")
   108→      .then((r) => r.json())
   109→      .then((data) => {
   110→        // API returns { items, unread_count } — handle both shapes defensively
   111→        const items = Array.isArray(data) ? data : (data?.items ?? []);
   112→        setNotifications(items.slice(0, 10));
   113→        setUnreadTotal(typeof data?.unread_count === "number" ? data.unread_count : items.length);
   114→      })
   115→      .catch(() => {});
   116→  }, [user]);
   117→
   118→  // Load once on mount / user change. The 30-second polling that previously
   119→  // lived here has been replaced by the WebSocket subscription below — the
   120→  // initial fetch is kept so a freshly-loaded tab shows the user's existing
   121→  // unread notifications without waiting for the next WS event.
   122→  React.useEffect(() => {
   123→    loadNotifications();
   124→  }, [loadNotifications]);
   125→
   126→  // ── D-4: WebSocket real-time subscription ────────────────────────────────
   127→  // Replaces the 30s polling. Handlers are recreated on every render (cheap)
   128→  // but the hook internally keeps the socket subscription stable across
   129→  // renders via a ref (see `src/hooks/use-realtime.ts`).
   130→  const flashBell = React.useCallback(() => {
   131→    setBellPulse(true);
   132→    if (pulseTimer.current) clearTimeout(pulseTimer.current);
   133→    pulseTimer.current = setTimeout(() => setBellPulse(false), 1_200);
   134→  }, []);
   135→
   136→  useRealtime({
   137→    "message:new": React.useCallback((data: any) => {
   138→      // Admin-sent message to portal — admin bell doesn't show portal
   139→      // notifications, but we flash the bell anyway so the admin gets
   140→      // visual confirmation that the send landed.
   141→      flashBell();
   142→      toast.message("Message sent", {
   143→        description: data?.preview ? String(data.preview).slice(0, 80) : undefined,
   144→      });
   145→    }, [flashBell]),
   146→
   147→    "offer:updated": React.useCallback((data: any) => {
   148→      // A status change on any offer in this tenant — flash + toast.
   149→      setUnreadTotal((c) => c + 1);
   150→      setNotifications((prev) => [{
   151→        id: `rt:offer:${data?.offerId ?? ""}:${Date.now()}`,
   152→        title: `Offer ${data?.offerNumber || ""} → ${data?.newStatus || "updated"}`,
   153→        message: data?.offerNumber
   154→          ? `Status changed from ${data?.oldStatus || "—"} to ${data?.newStatus || "—"}.`
   155→          : "Offer status updated.",
   156→        type: `offer_${data?.newStatus || "updated"}`,
   157→        read: false,
   158→        created_at: new Date().toISOString(),
   159→        entity_type: "offer",
   160→      }, ...prev].slice(0, 10));
   161→      flashBell();
   162→      const isNewAccepted = String(data?.newStatus || "").toLowerCase() === "accepted";
   163→      const isNewRejected = String(data?.newStatus || "").toLowerCase() === "rejected";
   164→      if (isNewAccepted) {
   165→        toast.success(`Offer ${data?.offerNumber || ""} accepted`);
   166→      } else if (isNewRejected) {
   167→        toast.error(`Offer ${data?.offerNumber || ""} ${data?.newStatus}`);
   168→      } else {
   169→        toast.message(`Offer ${data?.offerNumber || ""} → ${data?.newStatus || "updated"}`);
   170→      }
   171→    }, [flashBell]),
   172→
   173→    "invoice:paid": React.useCallback((data: any) => {
   174→      // A payment was recorded against an invoice in this tenant.
   175→      setUnreadTotal((c) => c + 1);
   176→      setNotifications((prev) => [{
   177→        id: `rt:invoice:${data?.invoiceId ?? ""}:${Date.now()}`,
   178→        title: data?.isFullPayment
   179→          ? `Invoice ${data?.invoiceNumber || ""} paid`
   180→          : `Partial payment on ${data?.invoiceNumber || ""}`,
   181→        message: data?.invoiceNumber
   182→          ? `${data?.isFullPayment ? "Paid in full" : "Partial payment"} — ${data?.method || ""}${data?.reference ? ` (ref: ${data.reference})` : ""}.`
   183→          : "Payment recorded.",
   184→        type: data?.isFullPayment ? "invoice_paid" : "invoice_partial",
   185→        read: false,
   186→        created_at: new Date().toISOString(),
   187→        entity_type: "invoice",
   188→      }, ...prev].slice(0, 10));
   189→      flashBell();
   190→      toast.success(
   191→        data?.isFullPayment
   192→          ? `Invoice ${data?.invoiceNumber || ""} paid`
   193→          : `Partial payment recorded`,
   194→        { description: data?.method ? `via ${data.method}` : undefined }
   195→      );
   196→    }, [flashBell]),
   197→
   198→    "portal:activity": React.useCallback((data: any) => {
   199→      // Something happened on the portal side (RFQ submitted, etc.).
   200→      setUnreadTotal((c) => c + 1);
   201→      const label =
   202→        data?.type === "rfq" ? `New RFQ: ${data?.productName || "product"}` :
   203→        data?.type === "portal_message" ? "Portal message" :
   204→        "Portal activity";
   205→      setNotifications((prev) => [{
   206→        id: `rt:portal:${data?.type ?? ""}:${data?.rfqId ?? ""}:${Date.now()}`,
   207→        title: label,
   208→        message: data?.partnerName ? `From ${data.partnerName}.` : "",
   209→        type: `portal_${data?.type || "activity"}`,
   210→        read: false,
   211→        created_at: new Date().toISOString(),
   212→        entity_type: data?.type === "rfq" ? "portal_rfq" : "portal_access",
   213→      }, ...prev].slice(0, 10));
   214→      flashBell();
   215→      toast.message(label, {
   216→        description: data?.partnerName ? `From ${data.partnerName}` : undefined,
   217→      });
   218→    }, [flashBell]),
   219→  },
   220→  // P0 / task D-FIX: when the WebSocket gateway (mini-services/notifications-service)
   221→  // is not deployed, the useRealtime hook falls back to 30s polling. Passing
   222→  // `loadNotifications` here means the polling tick re-fetches /api/notifications
   223→  // (same path the initial-mount load uses), so the bell stays fresh even
   224→  // without the live WS push. When the mini-service IS deployed, the socket
   225→  // connects, polling clears, and the live WS handlers take over.
   226→  loadNotifications,
   227→);
   228→
   229→  // Cleanup the pulse timer on unmount.
   230→  React.useEffect(() => () => {
   231→    if (pulseTimer.current) clearTimeout(pulseTimer.current);
   232→  }, []);
   233→
   234→  const unreadCount = unreadTotal;
   235→
   236→  const markAllRead = async () => {
   237→    try {
   238→      const res = await fetch("/api/notifications?markAllRead=true", { method: "PUT" });
   239→      if (res.ok) {
   240→        setNotifications((prev) => prev.map((n) => ({ ...n, read: true })));
   241→        setUnreadTotal(0);
   242→        toast.success("All notifications marked as read");
   243→      }
   244→    } catch {
   245→      // silently fail
   246→    }
   247→  };
   248→
   249→  function handleNotifClick(n: NotifItem) {
   250→    setNotifications((prev) => prev.map((item) => item.id === n.id ? { ...item, read: true } : item));
   251→    if (!n.read) setUnreadTotal((c) => Math.max(0, c - 1));
   252→    fetch(`/api/notifications/${n.id}`, {
   253→      method: "PUT",
   254→      headers: { "Content-Type": "application/json" },
   255→      body: JSON.stringify({ read: true }),
   256→    }).catch(() => {});
   257→    if (n.entity_type) {
   258→      const viewMap: Record<string, string> = {
   259→        offer: "offers", invoice: "invoices", proforma: "proformas",
   260→        kyc_submission: "kyc-review", deal: "deals", task: "tasks",
   261→        portal_rfq: "portal-rfqs", document: "documents",
   262→        portal_access: "portal-rfqs", partner: "partners", product: "products",
   263→      };
   264→      const targetView = viewMap[n.entity_type];
   265→      if (targetView) setView(targetView as any);
   266→    }
   267→    setNotifOpen(false);
   268→  }
   269→
   270→  async function logout() {
   271→    await fetch("/api/auth/logout", { method: "POST" });
   272→    // D-4: tear down the WebSocket so a logged-out tab doesn't keep an open
   273→    // connection under a stale identity. The next login will reconnect with
   274→    // the new user's rooms.
   275→    disconnectRealtime();
   276→    setUser(null);
   277→    setView("dashboard");
   278→    useI18nStore.getState().reset();
   279→    toast.success("Signed out.");
   280→  }
   281→
   282→  function openPortal() {
   283→    setAppMode("portal");
   284→  }
   285→
   286→  const translatedViewTitle = t(locale, view);
   287→  const viewTitle = translatedViewTitle === view ? VIEW_TITLE_FALLBACK : translatedViewTitle;
   288→
   289→  return (
   290→    <>
   291→      {/* ── Left: Breadcrumb-style view title ── */}
   292→      <div className="min-w-0 flex-1 flex items-center gap-2.5">
   293→        {/* Breadcrumb prefix — desktop only; on mobile the view title alone
   294→            is enough context and every pixel of width matters. */}
   295→        <div className="hidden sm:flex items-center gap-1.5 text-muted-foreground/60 text-xs font-medium tracking-wide uppercase shrink-0">
   296→          <PanelRight className="size-3.5" />
   297→          <span>VELOS</span>
   298→          <span className="text-muted-foreground/30">/</span>
   299→        </div>
   300→        <h2 className="text-[15px] font-semibold tracking-tight text-foreground truncate smooth min-w-0 shrink">
   301→          {viewTitle}
   302→        </h2>
   303→
   304→        {/* Tenant Context Switcher — shrinks to icon-only on mobile */}
   305→        <div className="min-w-0 shrink-0">
   306→          <TenantContextSwitcher />
   307→        </div>
   308→      </div>
   309→
   310→      {/* ── Right: Action cluster ── */}
   311→      <div className="flex items-center gap-0.5 sm:gap-1 shrink-0">
   312→        {/* Search — opens command palette */}
   313→        <Tooltip>
   314→          <TooltipTrigger asChild>
   315→            <Button
   316→              variant="ghost"
   317→              size="icon"
   318→              onClick={() => searchOpen()}
   319→              className="size-8 rounded-lg text-muted-foreground hover:text-foreground hover:bg-muted/50 smooth"
   320→            >
   321→              <Search className="size-4" />
   322→            </Button>
   323→          </TooltipTrigger>
   324→          <TooltipContent side="bottom" className="text-xs">
   325→            {t(locale, "search")} <kbd className="ml-1 text-[10px] bg-muted px-1 py-0.5 rounded">⌘K</kbd>
   326→          </TooltipContent>
   327→        </Tooltip>
   328→
   329→        {/* Real-time Notifications */}
   330→        <Popover open={notifOpen} onOpenChange={setNotifOpen}>
   331→          <PopoverTrigger asChild>
   332→            <Button
   333→              variant="ghost"
   334→              size="icon"
   335→              className={cn(
   336→                "relative size-8 rounded-lg text-muted-foreground hover:text-foreground hover:bg-muted/50 smooth",
   337→                bellPulse && "animate-pulse",
   338→              )}
   339→            >
   340→              <Bell className="size-4" />
   341→              {/* D-4: brief ping ring when a new WS notification arrives.
   342→                  Renders ABOVE the unread-count badge so the two don't
   343→                  collide visually. */}
   344→              {bellPulse && (
   345→                <span className="absolute inset-0 rounded-lg ring-2 ring-emerald-500/70 animate-ping" aria-hidden />
   346→              )}
   347→              {unreadCount > 0 && (
   348→                <span className="absolute -top-0.5 -right-0.5 flex items-center justify-center min-w-[16px] h-[16px] px-1 rounded-full bg-emerald-500 text-white text-[9px] font-semibold leading-none ring-2 ring-background">
   349→                  {unreadCount}
   350→                </span>
   351→              )}
   352→            </Button>
   353→          </PopoverTrigger>
   354→          <PopoverContent
   355→            align="end"
   356→            sideOffset={8}
   357→            style={{ maxHeight: "min(80vh, var(--radix-popover-content-available-height, 80vh))" }}
   358→            className="w-[min(92vw,380px)] p-0 rounded-xl border border-border/60 shadow-soft-lg flex flex-col overflow-hidden"
   359→          >
   360→            {/* Header */}
   361→            <div className="flex items-center justify-between gap-2 px-4 py-3 border-b border-border/60 shrink-0">
   362→              <div className="flex items-center gap-2 min-w-0">
   363→                <h3 className="text-sm font-semibold truncate">{t(locale, "notifications")}</h3>
   364→                {unreadCount > 0 && (
   365→                  <span className="inline-flex items-center justify-center min-w-[18px] h-[18px] px-1 rounded-full bg-emerald-500/15 text-emerald-700 dark:text-emerald-300 text-[10px] font-semibold tabular">
   366→                    {unreadCount}
   367→                  </span>
   368→                )}
   369→              </div>
   370→              <Button
   371→                variant="ghost"
   372→                size="sm"
   373→                className="h-7 px-2 text-xs text-muted-foreground hover:text-foreground"
   374→                disabled={unreadCount === 0}
   375→                onClick={markAllRead}
   376→              >
   377→                <CheckCheck className="size-3.5 mr-1" />
   378→                {t(locale, "mark-all-read")}
   379→              </Button>
   380→            </div>
   381→
   382→            {/* Body — flexes to fill whatever space remains between the
   383→                header and footer, so the footer ("View all") always stays
   384→                visible and clickable instead of being pushed off-screen.
   385→                Plain overflow-y-auto instead of the Radix ScrollArea — its
   386→                Viewport doesn't reliably inherit a flex-computed height
   387→                (measured: Viewport grew to full list content height and
   388→                spilled past the popover boundary instead of scrolling). */}
   389→            <div className="flex-1 min-h-0 overflow-y-auto custom-scroll">
   390→              {notifications.length === 0 ? (
   391→                <div className="py-12 px-4 flex flex-col items-center text-center">
   392→                  <div className="size-10 rounded-full bg-muted/60 flex items-center justify-center mb-2">
   393→                    <Bell className="size-5 text-muted-foreground" />
   394→                  </div>
   395→                  <p className="text-sm font-medium">{t(locale, "no-notifications")}</p>
   396→                  <p className="text-xs text-muted-foreground mt-0.5">{t(locale, "all-caught-up")}</p>
   397→                </div>
   398→              ) : (
   399→                <ul className="divide-y divide-border/50">
   400→                  {notifications.map((n) => (
   401→                    <li key={n.id}>
   402→                      <button
   403→                        type="button"
   404→                        onClick={() => handleNotifClick(n)}
   405→                        className={cn(
   406→                          "w-full text-left px-4 py-3 flex items-start gap-3 smooth relative group",
   407→                          !n.read && "bg-accent/30",
   408→                          "hover:bg-accent/50 focus-visible:bg-accent/50",
   409→                        )}
   410→                      >
   411→                        {!n.read && (
   412→                          <span className="absolute left-1.5 top-1/2 -translate-y-1/2 size-1.5 rounded-full bg-emerald-500" />
   413→                        )}
   414→                        <div className={cn(
   415→                          "size-8 shrink-0 rounded-full bg-muted/60 flex items-center justify-center",
   416→                          n.type.includes("overdue") || n.type.includes("rejected") ? "text-destructive" :
   417→                          n.type.includes("accepted") || n.type.includes("won") || n.type.includes("paid") ? "text-emerald-600 dark:text-emerald-400" :
   418→                          "text-muted-foreground"
   419→                        )}>
   420→                          {n.type.includes("overdue") || n.type.includes("rejected") ? <AlertTriangle className="size-4" /> :
   421→                           n.type.includes("accepted") || n.type.includes("won") || n.type.includes("paid") ? <Check className="size-4" /> :
   422→                           n.type.includes("kyc") ? <ShieldCheck className="size-4" /> :
   423→                           n.type.includes("offer") ? <FileText className="size-4" /> :
   424→                           n.type.includes("task") ? <Clock className="size-4" /> :
   425→                           n.type.includes("stock") ? <Package className="size-4" /> :
   426→                           <Info className="size-4" />}
   427→                        </div>
   428→                        <div className="flex-1 min-w-0">
   429→                          <div className="flex items-baseline gap-2">
   430→                            <p className={cn("text-sm truncate", n.read ? "font-medium" : "font-semibold")}>
   431→                              {n.title}
   432→                            </p>
   433→                            <span className="text-[10px] text-muted-foreground ml-auto shrink-0 tabular">
   434→                              {fmtRelative(n.created_at)}
   435→                            </span>
   436→                          </div>
   437→                          {n.message && (
   438→                            <p className="text-xs text-muted-foreground line-clamp-2 mt-0.5">{n.message}</p>
   439→                          )}
   440→                        </div>
   441→                      </button>
   442→                    </li>
   443→                  ))}
   444→                </ul>
   445→              )}
   446→            </div>
   447→
   448→            <Separator className="shrink-0" />
   449→            <div className="p-2 shrink-0">
   450→              <Button
   451→                variant="ghost"
   452→                size="sm"
   453→                className="w-full justify-center text-xs text-muted-foreground hover:text-foreground"
   454→                onClick={() => { setView("security"); setNotifOpen(false); }}
   455→              >
   456→                {t(locale, "view-all")}
   457→              </Button>
   458→            </div>
   459→          </PopoverContent>
   460→        </Popover>
   461→
   462→        {/* Language Switcher */}
   463→        <DropdownMenu>
   464→          <Tooltip>
   465→            <TooltipTrigger asChild>
   466→              <DropdownMenuTrigger asChild>
   467→                <Button
   468→                  variant="ghost"
   469→                  size="icon"
   470→                  className="size-8 rounded-lg text-muted-foreground hover:text-foreground hover:bg-muted/50 smooth"
   471→                >
   472→                  <Globe className="size-4" />
   473→                </Button>
   474→              </DropdownMenuTrigger>
   475→            </TooltipTrigger>
   476→            <TooltipContent side="bottom" className="text-xs">
   477→              {t(locale, "switch-language")}
   478→            </TooltipContent>
   479→          </Tooltip>
   480→          <DropdownMenuContent align="end" className="w-44 rounded-xl p-1.5 shadow-soft-lg border-border/50">
   481→            {(Object.keys(LOCALE_LABELS) as Locale[]).map((loc) => (
   482→              <DropdownMenuItem
   483→                key={loc}
   484→                onClick={() => setLocale(loc)}
   485→                className={cn("cursor-pointer rounded-lg px-3 py-2 text-sm smooth", locale === loc && "bg-accent/50")}
   486→              >
   487→                <span className="mr-2">{LOCALE_FLAGS[loc]}</span>
   488→                {LOCALE_LABELS[loc]}
   489→                {locale === loc && <Check className="size-3.5 ml-auto text-primary" />}
   490→              </DropdownMenuItem>
   491→            ))}
   492→          </DropdownMenuContent>
   493→        </DropdownMenu>
   494→
   495→        {/* Theme Customization */}
   496→        <DropdownMenu>
